def cli(arguments):
    return subprocess.run(
        ["uv", "run", "--extra", "cloud", "modal", *arguments, "--profile", "amandogra2016"],
        capture_output=True,
        text=True,
        check=True,
    )


def cleanup():
    if not json.loads((ROOT / "final-audit.json").read_text())["valid"]:
        raise ValueError("Final audit is incomplete")
    releases = json.loads((ROOT / "huggingface-release.json").read_text())
    if set(releases) != {"control", "treatment"}:
        raise ValueError("Private selected release inventory differs")
    for arm, receipt in releases.items():
        checked = receipt["remote_download_verification"]
        if not checked["verified"] or checked["device"] != "cpu" or checked["inputs"] != 8:
            raise ValueError("Fresh private release verification is incomplete")
        local = Path("artifacts/v6-releases") / arm
        fresh = Path(receipt["verified_download_directory"])
        for directory in (local, fresh):
            release_inventory(directory)
            if (
                file_hash(directory / "artifact-manifest.json")
                != receipt["artifact_manifest_sha256"]
            ):
                raise ValueError("Verified selected release manifest differs")
    retained = json.loads((ROOT / "retained-predictions.json").read_text())
    for name, digest in retained.items():
        if file_hash(Path(name)) != digest:
            raise ValueError("Retained prediction checksum differs")
    ownership = json.loads((ROOT / "modal-ownership.json").read_text())
    if ownership["volume"] != "decision-model-lab-v6-runs":
        raise ValueError("Unexpected owned Volume")
    volume = modal.Volume.from_name(ownership["volume"], create_if_missing=False)
    volume.hydrate()
    if volume.object_id != ownership["volume_id"]:
        raise ValueError("Owned Volume ID differs")
    for app_id in ownership["app_ids"]:
        if not app_id.startswith("ap-"):
            raise ValueError("Invalid owned app ID")
    stopped = inspect_owned_apps(ownership["app_ids"])
    receipts = [
        {
            "app_id": row["app_id"],
            "action": "verified already stopped; no stop requested",
            "lifecycle": row,
        }
        for row in stopped
    ]
    deleted = cli(["volume", "delete", ownership["volume"], "--yes"])
    actions = {"stop_receipts": receipts, "volume_delete_stdout": deleted.stdout}
    atomic_json(ROOT / "cleanup-actions.json", actions)
    volumes = json.loads(cli(["volume", "list", "--json"]).stdout)
    if any(row["name"] == ownership["volume"] for row in volumes):
        raise ValueError("Owned Volume still exists")
    result = {
        "checked_at_utc": datetime.now(UTC).isoformat(),
        "owned_volume_absent": True,
        "apps": inspect_owned_apps(ownership["app_ids"]),
        "actions": actions,
        "zero_owned_tasks": True,
        "unrelated_resources_modified": False,
    }
    atomic_json(ROOT / "cleanup.json", result)
    return result


def costs():
    if any(
        (ROOT / name).exists()
        for name in (
            "billing-final-resources.json",
            "billing-final.json",
            "billing-attribution.json",
        )
    ):
        raise FileExistsError("Preserve the prior billing observation before taking another")
    start = datetime.fromisoformat(
        json.loads((ROOT / "preflight.json").read_text())["snapshot_utc"]
    )
    resolution = "h" if (datetime.now(UTC) - start).total_seconds() <= 7 * 86400 else "d"
    end = datetime.now(UTC).date() + timedelta(days=1)
    rows = json.loads(
        cli(
            [
                "billing",
                "report",
                "--start",
                start.date().isoformat(),
                "--end",
                end.isoformat(),
                "--resolution",
                resolution,
                "--show-resources",
                "--json",
            ]
        ).stdout
    )
    summary = json.loads(cli(["billing", "summary", "--json"]).stdout)
    atomic_json(ROOT / "billing-final-resources.json", rows)
    atomic_json(ROOT / "billing-final.json", summary)
    ownership = json.loads((ROOT / "modal-ownership.json").read_text())
    ids = set(ownership["app_ids"]) | {ownership["volume_id"]}
    selected = [r for r in rows if r["object_id"] in ids]
    grouped = defaultdict(Decimal)
    for row in selected:
        grouped[row["resource"]] += Decimal(row["cost"])
    result = {
        "snapshot_utc": datetime.now(UTC).isoformat(),
        "owned_resource_metered_usd": str(sum(grouped.values(), Decimal(0))),
        "resources_usd": {k: str(v) for k, v in grouped.items()},
        "workspace_billed_usd": summary["billed_cost"],
        "scoped_rows": selected,
        "scope": "Owned V6 app/Volume rows only; observed snapshot, not a final invoice.",
        "requested_interval_utc": {"start": start.date().isoformat(), "end": end.isoformat()},
        "resolution": resolution,
        "interval_note": (
            "Explicit complete-day range includes the current UTC hour; metering may lag."
        ),
        "raw_report_sha256": file_hash(ROOT / "billing-final-resources.json"),
    }
    atomic_json(ROOT / "billing-attribution.json", result)
    return {k: v for k, v in result.items() if k != "scoped_rows"}


def audit():
    torch.set_num_threads(2)
    protocol = verify(ROOT)
    spec = protocol["settings"]
    frozen = check_seal(json.loads((ROOT / "selection-freeze.json").read_text()))
    evaluation = json.loads((ROOT / "evaluation.json").read_text())
    if (evaluation["protocol_identity"], evaluation["selection_identity"]) != (
        protocol["identity"],
        frozen["identity"],
    ):
        raise ValueError("Evaluation identities differ")
    manifest = json.loads((ROOT / "data/manifest.json").read_text())
    recovery = json.loads((ROOT / "checkpoint-recovery.json").read_text())
    if (
        not recovery["checkpoint_hashes_verified"]
        or recovery["optimizer_and_rng_restoration_pending"]
    ):
        raise ValueError("Saved checkpoint recovery is incomplete")
    for name, digest in recovery["retained_partial_candidate_files"].items():
        if file_hash(ROOT / recovery["partial_candidate_preserved_at"] / name) != digest:
            raise ValueError("Original partial candidate asset differs")
    recovered_candidate = json.loads((ROOT / "candidate-records/control-step250.json").read_text())
    if (
        recovered_candidate["weights_sha256"]
        != recovery["checkpoint_marker"]["files"]["model.safetensors"]
    ):
        raise ValueError("Recovered update-250 candidate differs from the original checkpoint")
    for arm in ("control", "treatment"):
        state = json.loads((ROOT / f"state-{arm}.json").read_text())
        trajectory = json.loads((ROOT / f"trajectory-{arm}.json").read_text())
        expected = manifest["splits"][f"train_{arm}"]["questions"] * spec["epochs"]
        if (
            not state["complete"]
            or state["presentations"] != expected
            or state["step"] != spec["candidate_steps"][-1]
            or state["protocol_identity"] != protocol["identity"]
        ):
            raise ValueError("Training exposure differs")
        updates = trajectory["updates"]
        if arm == "control":
            original = json.loads((ROOT / "failed-control-step250-progress.json").read_text())
            if updates[:250] != original["updates"]:
                raise ValueError("The original 250-update trajectory changed during recovery")
        if [r["step"] for r in updates] != list(range(1, state["step"] + 1)) or sum(
            r["examples"] for r in updates
        ) != expected:
            raise ValueError("Training trajectory differs")
        tokens = (
            protocol["audit"]["token_audit"][f"train_{arm}"]["total_input_tokens"] * spec["epochs"]
        )
        if sum(r["input_tokens"] for r in updates) != tokens:
            raise ValueError("Token exposure differs")
    retained = json.loads((ROOT / "retained-records.json").read_text())
    for name, digest in retained.items():
        if file_hash(ROOT / name) != digest:
            raise ValueError(f"Retained record differs: {name}")
    total, hashes = 0, {}
    expected_predictions = {
        ROOT / "predictions" / name / f"{split}.jsonl"
        for name, files in frozen["prediction_files"].items()
        for split in files
    }
    expected_predictions |= {
        ROOT / "predictions" / arm / f"{split}.jsonl"
        for arm in ("control", "treatment")
        for split in TESTS
    }
    expected_predictions |= {
        ROOT / "predictions" / "parent" / f"{split}.jsonl"
        for split in ("test_multi", "test_original", "test_counterfactual")
    }
    if set((ROOT / "predictions").rglob("*.jsonl")) != expected_predictions:
        raise ValueError("Prediction file inventory differs from the frozen scope")
    expected_total = sum(
        manifest["splits"][path.stem]["questions"] for path in expected_predictions
    )
    for path in sorted((ROOT / "predictions").rglob("*.jsonl")):
        name, split = path.parent.name, path.stem
        rows, logits, labels = predictions(ROOT, name, split)
        total += len(rows)
        hashes[str(path)] = file_hash(path)
        if name in frozen["prediction_files"]:
            if (
                split in frozen["prediction_files"][name]
                and file_hash(path) != frozen["prediction_files"][name][split]
            ):
                raise ValueError("Selection prediction checksum differs")
        if name in ("control", "treatment") and split in TESTS:
            if file_hash(path) != evaluation["arms"][name]["prediction_files"][split]:
                raise ValueError("Final prediction checksum differs")
            selected = frozen["choices"][name]["calibrated_nll"]
            rebuilt = metrics(
                rows,
                {"logits": logits, "labels": labels},
                frozen["calibrations"][selected]["temperature"],
                spec,
            )
            if rebuilt != evaluation["arms"][name]["splits"][split]:
                raise ValueError(f"Recomputed metrics differ: {name}/{split}")
    if total != expected_total:
        raise ValueError("Raw prediction row count differs from the frozen scope")
    storage = json.loads((ROOT / "checkpoint-storage.json").read_text())
    if (
        storage["candidates_verified"] != 10
        or not storage["all_additional_weight_metadata_verified"]
    ):
        raise ValueError("Candidate retention is incomplete")
    for record in json.loads((ROOT / "retained-candidates.json").read_text()).values():
        if record["path"] is not None and file_hash(Path(record["path"])) != record["sha256"]:
            raise ValueError("Retained selected/initial candidate differs")
    for arm in ("control", "treatment"):
        release = release_inventory(Path("artifacts/v6-releases") / arm)
        if release["protocol_identity"] != protocol["identity"]:
            raise ValueError("Selected release protocol differs")
    old = json.loads(
        Path("artifacts/v4-source-snapshot/manifest-final-with-reports.json").read_text()
    )["files"]
    unchanged = 0
    for name, digest in old.items():
        if name.startswith(("src/", "v4_lab/", "tests/", "configs/")) or name in (
            "pyproject.toml",
            "uv.lock",
        ):
            if file_hash(Path(name)) != digest:
                raise ValueError(f"Historical scientific/build file changed: {name}")
            unchanged += 1
    atomic_json(ROOT / "retained-predictions.json", hashes)
    result = {
        "valid": True,
        "protocol_identity": protocol["identity"],
        "prediction_rows_verified": total,
        "prediction_rows_expected": expected_total,
        "prediction_files_verified": len(expected_predictions),
        "serialized_data_unchanged_from_pre_execution_audit": True,
        "historical_scientific_build_files_unchanged": unchanged,
        "candidate_count_verified": 10,
        "analysis_sha256": file_hash(ROOT / "analysis.json"),
    }
    atomic_json(ROOT / "final-audit.json", result)
    return result


def report():
    import matplotlib

    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

    destination = Path("reports/V6_RESULTS.md")
    if destination.exists():
        raise FileExistsError(destination)
    evaluation = json.loads((ROOT / "evaluation.json").read_text())
    analysis = json.loads((ROOT / "analysis.json").read_text())
    frozen = json.loads((ROOT / "selection-freeze.json").read_text())
    primary = analysis["primary"]
    protocol = verify(ROOT)
    tokens = {
        arm: protocol["audit"]["token_audit"][f"train_{arm}"]["total_input_tokens"]
        * protocol["settings"]["epochs"]
        for arm in ("control", "treatment")
    }
    lines = [
        "# V6: matched multi-entity training results",
        "",
        "Single-seed continuation from the same pinned V4 mixed head; "
        "36,288 presentations and 1,134 updates per arm.",
        "",
        f"Input-token exposure: control {tokens['control']:,}; "
        f"treatment {tokens['treatment']:,} "
        f"({tokens['treatment'] / tokens['control'] - 1:.2%} more). "
        "The final trajectory audit reconciles these totals against every update.",
        "",
        f"Frozen protocol: `{evaluation['protocol_identity']}`. "
        f"Selection identity: `{evaluation['selection_identity']}`.",
        "",
        "Primary complete-world treatment-minus-control gain: "
        f"**{100 * primary['estimate']:.2f} percentage points**, "
        f"95% blocked bootstrap interval **[{100 * primary['interval'][0]:.2f}, "
        f"{100 * primary['interval'][1]:.2f}] points** across {primary['families']} worlds "
        f"and {primary['blocks']} label-rotation blocks.",
        "",
        "The primary test varies label composition. Unlike V5, a constant class can correctly "
        "answer some homogeneous worlds. Compare the arms on these fresh, paired inputs; "
        "their complete-world rates are not directly comparable "
        "with V5's one-of-each construction.",
        "",
        "## Selected checkpoints",
        "",
        "| Arm | Update | Calibrated selection NLL | Final temperature |",
        "| --- | ---: | ---: | ---: |",
    ]
    for arm in ("control", "treatment"):
        name = frozen["choices"][arm]["calibrated_nll"]
        lines.append(
            f"| {arm} | {frozen['candidates'][name]['step']} | "
            f"{frozen['selection_scores'][name]['calibrated_nll']:.6f} | "
            f"{frozen['calibrations'][name]['temperature']:.6f} |"
        )
    lines += [
        "",
        "## Fresh tests",
        "",
        "| Test | Control accuracy | Treatment accuracy | "
        "Control complete families | Treatment complete families |",
        "| --- | ---: | ---: | ---: | ---: |",
    ]
    for split in TESTS:
        control, treatment = (
            evaluation["arms"][arm]["splits"][split] for arm in ("control", "treatment")
        )
        lines.append(
            f"| {split} | {control['raw']['accuracy']:.2%} | {treatment['raw']['accuracy']:.2%} | "
            f"{control['all_questions_in_family_correct']:.2%} | "
            f"{treatment['all_questions_in_family_correct']:.2%} |"
        )
    lines += [
        "",
        "Complete families for transformations require every included variant and entity "
        "query to be correct; base complete worlds use only test_multi. "
        "Transfers have distinct generators and are descriptive.",
        "",
        "## World size",
        "",
        "| Entities | Control complete worlds | Treatment complete worlds |",
        "| --- | ---: | ---: |",
    ]
    for size in (3, 4, 5):
        counts = [
            analysis["complete_worlds_by_size"][arm][str(size)] for arm in ("control", "treatment")
        ]
        lines.append(
            f"| {size} | {counts[0]['correct']}/{counts[0]['families']} | "
            f"{counts[1]['correct']}/{counts[1]['families']} |"
        )
    lines += [
        "",
        "## Confidence and old behavior",
        "",
        "| Arm | Test | UNKNOWN accuracy | Wrong with raw confidence ≥90% | "
        "Wrong with final-calibrated confidence ≥90% |",
        "| --- | --- | ---: | ---: | ---: |",
    ]
    for arm in ("control", "treatment"):
        for split in (
            "test_multi",
            "test_original",
            "test_counterfactual",
            "test_multi_depth",
            "test_multi_wording",
            "test_multi_signed_chain",
        ):
            record = evaluation["arms"][arm]["splits"][split]
            lines.append(
                f"| {arm} | {split} | {record['unknown_accuracy']:.2%} | "
                f"{record['raw']['high_confidence_wrong']}/{record['raw']['questions']} | "
                f"{record['calibrated']['high_confidence_wrong']}/{record['raw']['questions']} |"
            )
    for split, result in analysis["retention_against_parent"].items():
        lines += [
            "",
            f"Treatment-minus-unchanged-parent UNKNOWN accuracy on {split}: "
            f"{result['left_minus_right']:.2%}, 95% parent-family interval "
            f"[{result['interval'][0]:.2%}, {result['interval'][1]:.2%}].",
        ]
    lines += [
        "",
        "Predeclared engineering checks: "
        + json.dumps(analysis["engineering_gate"], sort_keys=True)
        + ".",
        "",
        "UNKNOWN is a semantic label; temperature changes confidence, not the selected class. "
        "Full confusion matrices, NLL, Brier, ECE and class counts "
        "are retained in evaluation.json.",
        "",
        "## Entity changes and transformations",
        "",
        "| Arm | Pair | Expected response | Both answers correct |",
        "| --- | --- | ---: | ---: |",
    ]
    for arm, checks in analysis["entity_checks"].items():
        for name, record in checks.items():
            lines.append(
                f"| {arm} | {name} | {record['expected_response']}/{record['pairs']} | "
                f"{record['both_correct']}/{record['pairs']} |"
            )
    lines += [
        "",
        "Expected responses include label equivariance under negation/support edits and "
        "invariance under renaming or unrelated changes. "
        "High consistency can reflect two wrong answers.",
        "",
        "## Preselected example",
        "",
        "Family v6-test_multi-multi-3-0000-0 was selected before inference. "
        "It has a homogeneous base composition; it is retained as specified.",
        "",
    ]
    rows, _, _ = predictions(ROOT, "control", "test_multi_transformations")
    selected = [
        (i, row) for i, row in enumerate(rows) if row.family_id == "v6-test_multi-multi-3-0000-0"
    ]
    base = next(row for _, row in selected if "-base-" in row.example_id)
    lines += [
        "Facts: " + " ".join(base.facts),
        "",
        "Rules:",
        "",
        *[f"- {rule}" for rule in base.rules],
        "",
        "| Variant | Statement | Oracle | Control | Treatment | Treatment calibrated confidence |",
        "| --- | --- | --- | --- | --- | ---: |",
    ]
    logits = {
        arm: predictions(ROOT, arm, "test_multi_transformations")[1]
        for arm in ("control", "treatment")
    }
    temperature = frozen["calibrations"][frozen["choices"]["treatment"]["calibrated_nll"]][
        "temperature"
    ]
    labels = ["TRUE", "FALSE", "UNKNOWN"]
    for index, row in selected:
        if row.example_id.rsplit("-", 2)[-2] not in (
            "base",
            "support_edit",
            "support_edit_inverse",
        ):
            continue
        chosen = [labels[int(logits[arm][index].argmax())] for arm in ("control", "treatment")]
        confidence = float((logits["treatment"][index] / temperature).softmax(-1).max())
        lines.append(
            f"| {row.example_id.rsplit('-', 2)[-2]} | {row.question} | {row.label.value} | "
            f"{chosen[0]} | {chosen[1]} | {confidence:.2%} |"
        )
    lines += ["", "Changes to this preselected world's facts:", ""]
    for variant in ("support_edit", "support_edit_inverse"):
        changed = next(row for _, row in selected if row.example_id.rsplit("-", 2)[-2] == variant)
        removed = sorted(set(base.facts) - set(changed.facts))
        added = sorted(set(changed.facts) - set(base.facts))
        lines.append(f"- {variant}: removed {removed}; added {added}.")
    lines += [
        "",
        "## Limits and evidence",
        "",
        "This tests a curriculum change for one matched seed pair. Entity count, rule mixture "
        "and token exposure also differ; it does not identify an internal binding mechanism. "
        "Label-rotation blocks and world splits prevent recorded canonical overlap, "
        "not all abstract-skeleton overlap. Fillers/missing branch facts and templated "
        "wording can provide cues. The two oracle audits share parsing and grounding utilities. "
        "Secondary comparisons are descriptive without multiplicity adjustment; "
        "collapsed intervals are not population guarantees.",
        "",
        "Read [the protocol](../docs/V6_PROTOCOL.md), [the journal](../RESEARCH_LOG.md), "
        "and [compact evidence](v6-evidence/inventory.json). "
        "Raw data, logits and trajectories remain in runs/study-v6. "
        "Scheduled weights are retained through pinned private HF storage, "
        "with selected bundles local and privately verified. "
        "Costs and cleanup have separate timestamped receipts. "
        "No Git commit or push is part of this study.",
        "",
    ]
    destination.write_text("\n".join(lines))
    figures = Path("reports/v6-figures")
    figures.mkdir(exist_ok=True)
    fig, axis = plt.subplots(figsize=(8, 4), layout="constrained")
    for offset, arm in enumerate(("control", "treatment")):
        values = [
            analysis["complete_worlds_by_size"][arm][str(size)]["correct"]
            / analysis["complete_worlds_by_size"][arm][str(size)]["families"]
            for size in (3, 4, 5)
        ]
        selected = frozen["choices"][arm]["calibrated_nll"]
        step = frozen["candidates"][selected]["step"]
        axis.bar(
            [index + (offset - 0.5) * 0.3 for index in range(3)],
            values,
            width=0.3,
            label=f"{arm} (update {step})",
        )
    axis.set_xticks(range(3), ["3 entities", "4 entities", "5 entities"])
    axis.set_ylim(0, 1)
    axis.set_ylabel("Complete-world accuracy")
    axis.set_title("V6: fresh worlds, matched training arms")
    axis.legend()
    fig.savefig(figures / "complete-worlds.png", dpi=170)
    fig.savefig(figures / "complete-worlds.svg")
    plt.close(fig)
    return {"report": str(destination)}


def evidence():
    destination = Path("reports/v6-evidence")
    if destination.exists():
        raise FileExistsError(destination)
    destination.mkdir()
    for path in sorted(ROOT.glob("*.json")):
        if path.name == "billing-final-resources.json":
            continue
        shutil.copyfile(path, destination / path.name)
    for name in (
        "cpu-tests-final.log",
        "cpu-tests-amended.log",
        "cpu-tests-recovery-validated.log",
        "cpu-v6-initial.log",
        "pilot-cli-initial.log",
        "lint-initial.log",
        "training.log",
        "finish-initial.log",
        "source-before-recovery-initial.log",
        "lint-recovery-validated.log",
        "format-recovery-validated.log",
        "lock-recovery-validated.log",
        "lint-final.log",
        "format-final.log",
        "lock-final.log",
        "git-whitespace-final.log",
        "recovery.log",
        "finish-training-resumed.log",
        "finish-selection.log",
        "finish-test.log",
        "finish-audit.log",
        "finish-private-releases.log",
        "finish-cleanup.log",
        "finish-billing.log",
        "descriptive-edit-slices.log",
    ):
        shutil.copyfile(ROOT / name, destination / f"{name}.txt")
    shutil.copyfile(
        Path("artifacts/v6-source-snapshot/manifest-before-inference.json"),
        destination / "source-before-inference-manifest.json",
    )
    shutil.copyfile(
        Path("artifacts/v6-source-snapshot/manifest-before-recovery.json"),
        destination / "source-before-recovery-manifest.json",
    )
    for path in sorted((ROOT / "source-before-packaging-fix").rglob("*.py")):
        relative = path.relative_to(ROOT / "source-before-packaging-fix")
        shutil.copyfile(
            path, destination / ("original-" + str(relative).replace("/", "-") + ".txt")
        )
    atomic_json(
        destination / "inventory.json",
        {p.name: file_hash(p) for p in sorted(destination.iterdir())},
    )
    return {"compact_evidence_files": len(list(destination.iterdir()))}

    action = parser.parse_args().action

    if action == "analyse":

        result = analyse(ROOT)
        print(
            json.dumps(
                {"primary": result["primary"], "engineering_gate": result["engineering_gate"]}
            )
        )
    else:
        print(json.dumps({"action": action, "result": globals()[action]()}))
